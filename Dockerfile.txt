# Estágio de compilação
FROM node:20-alpine AS builder
WORKDIR /app

# Copia arquivos de dependência
COPY package*.json ./
RUN npm ci

# Copia todo o código fonte e compila
COPY . .
RUN npm run build

# Estágio de execução em produção
FROM node:20-alpine AS runner
WORKDIR /app

ENV NODE_ENV=production

# Instala apenas as dependências de produção necessárias em tempo de execução
COPY package*.json ./
RUN npm ci --omit=dev

# Copia a build do estágio anterior
COPY --from=builder /app/dist ./dist

# Railway configura e expõe automaticamente a porta via variável de ambiente PORT
EXPOSE 3000

CMD ["npm", "start"]
